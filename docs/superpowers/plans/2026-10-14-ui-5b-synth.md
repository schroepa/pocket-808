> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Der Synth (Bass/Lead) der alten App kommt in die neue Oberfläche, mit allen Funktionen aus dem Abschnitt „Synth (5b)“ der Funktionsliste. Er erscheint auf Mobile als Akkordeon, auf dem Tablet als Tab und auf dem Desktop im Bereich `synth`.

**Architecture:**
- **Reine Synth-Logik** liegt in `core/synth.ts`: Wert setzen, Wellenform, Oktave, Preset, Stummschalten, Notenabdeckung der Roll.
- **Oberfläche** liegt in `ui/panels/synth/`: Kopf, Piano-Roll, Module, Presets.
- **Neue Steuerelemente:** Schieberegler (für die ADSR) und Auswahlliste.
- **Gerät:** Der Synth wird über `synthDevice(c)` als erstes Gerät im Geräte-Halter registriert. Seine Zusammenfassung zeigt der Kopf (Mobile/Tablet).
- **Audio:** Die Oberfläche hört Noten über die neue Methode `UiAudio.auditionNote` vor.

**Tech Stack:** wie 5a (TypeScript 6, Vite 8, Vitest 5 mit jsdom, Playwright mit axe).

**Spec:** `docs/superpowers/specs/2026-10-14-ui-layouts-design.md` (Teilschritt 5b) und `docs/superpowers/specs/2026-10-14-ui-funktionsliste.md` (Abschnitt „Synth (5b)“). Verhaltensreferenz ist `legacy/index.html`, die Funktionen `buildSynth`, `renderSynth`, `renderRoll`, `placeNote`, `markRollNow`, `renderSum`, `envPath`, `addSlider` und `fmtT`.

## Review Focus

1. **Tippen in den Ausklang einer langen Note:** Eine Note ab Step 1 mit Länge 1/4 deckt die Steps 1–4 ab. Ein Tipp auf Step 3 derselben Stufe löscht die Note (Legacy `placeNote`) und hört nicht vor. Eine andere Stufe auf Step 3 setzt eine neue Note und kürzt die alte. Abgedeckt durch Task 1 (core) und Task 5 (Roll).
2. **Takthälfte 9–16:** Die Roll zeigt die Steps 9–16. Das Lauflicht erscheint nur, wenn der laufende Step auf der sichtbaren Hälfte liegt, und der Hälften-Chip leuchtet für die gespielte Hälfte. Abgedeckt in Task 5.
3. **Tonart oder Skala wechseln:** Die Zeilen werden neu beschriftet (Notennamen, `aria-label` der Zellen). Die Noten behalten ihre Stufen, die Grundton-Zeilen wechseln mit der Skalenlänge (Pentatonik: 5 Stufen). Abgedeckt in Task 4 und 5.
4. **Preset-Markierung:** Nach einem Preset lässt jeder Regler- oder Wellenformwechsel die Markierung verschwinden, die Oktave nicht. Beim Umschalten Bass ↔ Lead zeigen Regler, Wellenform, Oktave und Presets den anderen Part. Abgedeckt in Task 1, 6 und 7.
5. **Gerät über Layouts:** Auf Mobile klappt der Synth auf (`open.synth` bleibt nach Neuladen erhalten), auf dem Tablet ist er der Tab, auf dem Desktop steht er im Bereich `synth`. In keinem Layout gibt es horizontalen Scroll. Abgedeckt in Task 8 und 9 (E2E).

```ts
import { effectiveLength } from './music.ts';
import { clamp } from './persist.ts';
import type { ChannelId, NoteRow, Part, State, SynthKey, SynthPatch, SynthValues } from './types.ts';

/*
 * Synth-Bedienung als reine Funktionen (Legacy buildSynth): Regler und Wellenform heben die
 * Preset-Markierung auf, die Oktave nicht; Stumm ist der Mixer-Kanal des Parts.
 */

const withPatch = (s: State, part: Part, fn: (p: SynthPatch) => SynthPatch): State => ({
  ...s,
  synth: { ...s.synth, [part]: fn(s.synth[part]) },
});

export const setSynthValue = (s: State, part: Part, key: SynthKey, v: number): State =>
  withPatch(s, part, (p) => ({ ...p, [key]: v, preset: null }));

export const setWave = (s: State, part: Part, wave: number): State =>
  withPatch(s, part, (p) => ({ ...p, wave, preset: null }));

export const shiftOctave = (s: State, part: Part, d: number): State =>
  withPatch(s, part, (p) => ({ ...p, oct: clamp(p.oct + d, -2, 2) }));

export const applySynthPreset = (
  s: State,
  part: Part,
  preset: { name: string; v: Readonly<SynthValues> },
): State => withPatch(s, part, (p) => ({ ...p, ...preset.v, preset: preset.name }));

export const toggleMute = (s: State, id: ChannelId): State => ({
  ...s,
  mix: { ...s.mix, [id]: { ...s.mix[id], m: !s.mix[id].m } },
});

/** Die nächste Note an oder vor `step` bestimmt die Abdeckung (spätere Noten kürzen frühere). */
export function noteAt(row: NoteRow, step: number): { start: number; d: number } | null {
  for (let a = step; a >= 0; a--) {
    const n = row[a];
    if (n) return a + effectiveLength(row, a) > step ? { start: a, d: n.d } : null;
  }
  return null;
}
```

    () => o.value.value,
    (v) => {
      track.style.setProperty('--v', String(v));
      track.setAttribute('aria-valuenow', String(Math.round(v * 100)));
      track.setAttribute('aria-valuetext', o.format(v));
    },
  );
  const apply = (v: number): void => {
    const x = clamp(v, 0, 1);
    o.write(x);
    o.readout.flash(`${o.tag ? `${o.tag()} ` : ''}${o.aria} ${o.format(x)}`);
  };
  const fromY = (y: number): number => {
    const r = track.getBoundingClientRect();
    return 1 - (y - r.top - 6) / Math.max(1, r.height - 12);
  };
  let drag: { y: number; moved: boolean } | null = null;
  let lastTap = -Infinity;
  listen(scope, track, 'pointerdown', (e) => {
    e.preventDefault();
    try {
      track.setPointerCapture(e.pointerId);
    } catch {
      /* jsdom und ältere Browser */
    }
    drag = { y: e.clientY, moved: false };
  });
  listen(scope, track, 'pointermove', (e) => {
    if (!drag) return;
    if (Math.abs(e.clientY - drag.y) > 3) drag.moved = true;
    if (drag.moved) apply(fromY(e.clientY));
  });
  listen(scope, track, 'pointerup', (e) => {
    if (!drag) return;
    const moved = drag.moved;
    drag = null;
    if (!moved) {
      const t = now();
      if (t - lastTap < 320) {
        apply(o.def());
        lastTap = -Infinity;
      } else {
        apply(fromY(e.clientY));
        lastTap = t;
      }
    }
    o.onEnd?.();
  });
  listen(scope, track, 'pointercancel', () => {
    drag = null;
  });
  listen(scope, track, 'keydown', (e) => {
    const fn = KEYS[e.key];
    if (!fn) return;
    e.preventDefault();
    apply(fn(o.value.peek(), e.shiftKey ? 0.1 : 0.02));
  });
  return wrap;
}

import { degreeName } from '../../../core/music.ts';

import { noteAt } from '../../../core/synth.ts';
import { must } from '../../../core/util.ts';
import { computed, signal } from '../../../store/signals.ts';
import type { UiContext } from '../../context.ts';
import { chip } from '../../controls/chip.ts';
import { markColumn } from '../../controls/marker.ts';
import { bind, h } from '../../dom.ts';
import './roll.css';

const LENGTHS = [
  [1, '1/16'],
  [2, '1/8'],
  [4, '1/4'],
] as const;

/** Piano-Roll des gewählten Parts, eine Takthälfte (8 Steps) × 12 Stufen (Legacy renderRoll). */
export function pianoRoll(c: UiContext): HTMLElement {
  const { store, scope, audio } = c;
  const page = signal(0);
  const playStep = computed(() => audio.playhead.value.step);
  const tonality = store.select(
    (s) => ({ key: s.key, scale: s.scale }),
    (a, b) => a.key === b.key && a.scale === b.scale,
  );
  const scaleLen = computed(() => must(SCALES[tonality.value.scale], 'Skala').s.length);

  const pages = ['1–8', '9–16'].map((t, i) =>
    chip(scope, {
      label: t,
      ariaLabel: `Steps ${t}`,
      active: computed(() => page.value === i),
      onClick: () => {
        page.value = i;
      },
    }),
  );
  pages.forEach((b, i) => {
    bind(
      scope,
      () => playStep.value >= 0 && playStep.value >> 3 === i,
      (on) => {
        b.classList.toggle('now', on);
      },
    );
  });
  const lengths = LENGTHS.map(([n, t]) =>
    chip(scope, {
      label: t,
      ariaLabel: `Notenlänge ${t}`,
      active: store.select((s) => s.noteLen === n),
      onClick: () => {
        store.update((s) => ({ ...s, noteLen: n }));
      },
    }),
  );

  const headCells = Array.from({ length: 8 }, () => h('span'));
  headCells.forEach((sp, col) => {
    bind(
      scope,
      () => page.value * 8 + col + 1,
      (n) => {
        sp.textContent = String(n);
      },
    );
  });
  const columns: HTMLButtonElement[][] = Array.from({ length: 8 }, () => []);
  const rows = Array.from({ length: ROWS }, (_, i) => {
    const d = ROWS - 1 - i;
    const name = computed(() => degreeName(d, tonality.value));
    const lab = h('button', {
      class: 'rl',
      attrs: { type: 'button' },
      on: {
        click: () => {
          audio.auditionNote(store.state.peek().part, d, 1);
        },
      },
    });
    bind(
      scope,
      () => name.value,
      (n) => {
        lab.textContent = n;
        lab.setAttribute('aria-label', `Ton ${n} anhören`);
      },
    );
    const cells = Array.from({ length: 8 }, (_, col) => {
      const step = computed(() => page.value * 8 + col);
      const state = store.select((s) => {
        const k = noteAt(currentPattern(s).notes[s.part], step.value);
        return k?.d === d ? (k.start === step.value ? 'on' : 'tail') : '';
      });
      const b = h('button', {
        class: 'pc',
        attrs: { type: 'button' },
        on: {
          click: () => {
            const s = step.peek();
            store.placeNote(d, s);
            const now = store.state.peek();
            const n = currentPattern(now).notes[now.part][s];
            if (n?.d === d && !audio.playing.peek()) audio.auditionNote(now.part, d, now.noteLen);
          },
        },
      });
      bind(
        scope,
        () => `${state.value}|${step.value % 4 === 0 ? 'q' : ''}`,
        (v) => {
          const [st, q] = v.split('|');
          b.classList.toggle('on', st === 'on');
          b.classList.toggle('tail', st === 'tail');
          b.classList.toggle('q', q === 'q');
          b.setAttribute('aria-pressed', String(st === 'on'));
        },
      );
      bind(
        scope,

import { WAVES } from '../../../core/constants.ts';
import { synthDefaults } from '../../../core/defaults.ts';
import { attackTime, decayTime, releaseTime } from '../../../core/envelope.ts';
import { setSynthValue, setWave, shiftOctave } from '../../../core/synth.ts';
import type { SynthKey } from '../../../core/types.ts';
import { computed } from '../../../store/signals.ts';
import type { UiContext } from '../../context.ts';
import { chip } from '../../controls/chip.ts';
import { envelopeView } from '../../controls/envelope.ts';
import { knob } from '../../controls/knob.ts';
import { segment } from '../../controls/segment.ts';
import { slider } from '../../controls/slider.ts';
import { bind, h } from '../../dom.ts';
import { fmtTime, partName } from './format.ts';
import './modules.css';

const DEFAULTS = synthDefaults();
const ADSR = [
  ['a', 'A', 'Attack', (v: number) => fmtTime(attackTime(v))],
  ['d', 'D', 'Decay', (v: number) => fmtTime(decayTime(v))],
  ['s', 'S', 'Sustain', (v: number) => `${Math.round(v * 100)} %`],
  ['r', 'R', 'Release', (v: number) => fmtTime(releaseTime(v))],
] as const;

/** Die fünf SH-101-Module des gewählten Parts (Legacy buildSynth, Abschnitt Sound). */
export function synthModules(c: UiContext): HTMLElement {
  const { store, scope, audio, readout } = c;
  const part = store.select((s) => s.part);
  const patch = store.select((s) => s.synth[s.part]);
  const tag = (): string => partName(part.peek()).toUpperCase();
  const hear = (): void => {
    audio.auditionNote(part.peek(), 0, 2);
  };
  const value = (k: SynthKey) => computed(() => patch.value[k]);
  const write = (k: SynthKey) => (v: number) => {
    store.update((s) => setSynthValue(s, s.part, k, v));
  };
  const kn = (k: SynthKey, label: string) =>
    knob(scope, {
      label,
      value: value(k),
      write: write(k),
      def: () => DEFAULTS[part.peek()][k],
      format: (v) => String(Math.round(v * 100)),
      readout,
      tag,
      onEnd: hear,
    });
  const env = (pre: 'f' | 'a', name: string) => {
    const keys = ADSR.map(([c2]) => `${pre}${c2}` as SynthKey);
    const view = envelopeView(
      scope,
      computed(() => {
        const p = patch.value;
        const [a, d, s, r] = keys.map((k) => p[k]) as [number, number, number, number];
        return [attackTime(a), decayTime(d), s, releaseTime(r)] as const;
      }),
    );
    const sliders = ADSR.map(([c2, label, full, format]) => {
      const k = `${pre}${c2}` as SynthKey;
      return slider(scope, {
        label,
        aria: `${name} ${full}`,
        value: value(k),
        write: write(k),
        def: () => DEFAULTS[part.peek()][k],
        format,
        readout,
        tag,
        onEnd: hear,
      });
    });
    return h('div', { class: 'shm' }, [h('h4', { text: `${name}-Env` }), view, h('div', { class: 'shs' }, sliders)]);
  };
  const oct = h('span', { class: 'oct' });
  bind(
    scope,
    () => patch.value.oct,
    (o) => {
      oct.textContent = `Okt ${o > 0 ? '+' : ''}${o}`;
    },
  );
  const shift = (d: number) => () => {
    store.update((s) => shiftOctave(s, s.part, d));
    hear();
  };
  return h('div', { class: 'sh' }, [
    h('div', { class: 'shm w2' }, [
      h('h4', { text: 'Oszillator' }),
      h('div', { class: 'syn-row' }, [
        segment(scope, {
          label: 'Wellenform',
          options: WAVES.map(([name], i) => ({ value: String(i), label: name })),
          value: computed(() => String(patch.value.wave)),
          write: (w) => {
            store.update((s) => setWave(s, s.part, Number(w)));
            hear();
          },
        }),
        h('div', { class: 'oct-grp' }, [
          chip(scope, { label: '−', ariaLabel: 'Oktave tiefer', onClick: shift(-1) }),
          oct,
          chip(scope, { label: '+', ariaLabel: 'Oktave höher', onClick: shift(1) }),
        ]),
      ]),
      h('div', { class: 'shk' }, [kn('det', 'Detune'), kn('sub', 'Sub')]),
    ]),
    h('div', { class: 'shm w2' }, [
      h('h4', { text: 'Filter' }),
      h('div', { class: 'shk' }, [kn('cut', 'Cutoff'), kn('res', 'Reso'), kn('env', 'Env')]),
    ]),
    env('f', 'Filter'),
    env('a', 'Amp'),
    h('div', { class: 'shm w2' }, [
      h('h4', { text: 'Out' }),
      h('div', { class: 'shk' }, [kn('dly', 'Delay'), kn('lvl', 'Level')]),
    ]),
  ]);
}

import { SYN_PRESETS } from '../../../core/constants.ts';
import { applySynthPreset } from '../../../core/synth.ts';

import { chip } from '../../controls/chip.ts';

export function synthPresets(c: UiContext): HTMLElement {
  const { store, scope, audio } = c;
  const list = h('div', { class: 'chips syn-presets', attrs: { role: 'group', 'aria-label': 'Presets' } });

      inner.dispose();

          }),
        ),
      );

  );

    h('div', { class: 'chips syn-tools' }, [
      chip(scope, {
        label: 'Stimme leeren',
        onClick: () => {
          store.clearNotes();
        },
      }),
    ]),
  ]);
}
